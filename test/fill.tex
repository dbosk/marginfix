% Test for deferrable notes: a long commentary note (LT2) that doesn't fit
% in what is left of the margin, followed by short footnotes that do.  In
% mode warn, as in marginfix v1.4, LT2 takes the footnotes after it to the
% next page, and the warning says that it did; in mode defer LT2 moves alone
% and the footnotes fill the margin after LT1; in mode split LT2 is split
% and the footnotes follow its first part.  See prioritycheck.tex for how
% the modes are chosen and checked.  Needs three runs.
\documentclass[a4paper]{memoir}
\setlrmarginsandblock{2cm}{6cm}{*}
\setmarginnotes{1em}{4.5cm}{1em}
\checkandfixthelayout
\usepackage{marginfix}
\footnotesinmargin
\usepackage{lipsum}
\input{test/prioritycheck}
\expectmovedtrue
\begin{document}
\chapter{Test}
\ltnote{\lipsum*[1][1-14]} \lipsum[1]

\lipsum[2] \ltnote{\lipsum*[2][1-8]\par\lipsum*[3][1-10]\par
  \lipsum*[4][1-10]\par\lipsum*[5][1-10]}
Mark\fn{\lipsum[6][1]} \lipsum[3] Mark\fn{\lipsum[7][1]}
\lipsum[4] Mark\fn{\lipsum[8][1]}

\lipsum[5-6] \ltnote{\lipsum*[1][1-3]} \lipsum[7-12]
\end{document}
