% Test of notes beside bottom floats in the article class, with margin
% notes, footnotes at the foot of the page, and a table at the bottom.  In
% mode use (\besidemode) the room added beside the table must be exactly
% what LaTeX took from the text for it, as in test/floats.tex; the
% footnotes come between the text and the table.  Needs two runs.
\ifdefined\besidemode\else\def\besidemode{reserve}\fi
\documentclass[a4paper]{article}
\usepackage[textwidth=12cm,left=2cm,marginparwidth=5cm,marginparsep=5mm]{geometry}
\usepackage{marginfix}
\usepackage{lipsum}
\providecommand\marginnotebesidefloats[1]{}
\marginnotebesidefloats{\besidemode}
\makeatletter
\AtBeginDocument{%
  \ifdefined\MFX@bottomroom
    \let\ft@bottomroom\MFX@bottomroom
    \def\MFX@bottomroom{\ft@bottomroom\ft@checkroom}%
  \fi}
\def\ft@checkroom{%
  \if@tempswa\else
    \begingroup
      \@tempdima\dimexpr\@colht-\@colroom\relax
      \typeout{floatstest: page \thepage: room \the\dimen@,
        expected \the\@tempdima}%
      \ifdim\dimexpr\dimen@-\@tempdima\relax>.01pt
        \PackageError{floatstest}{Room \the\dimen@, expected
          \the\@tempdima}\@ehc\fi
      \ifdim\dimexpr\@tempdima-\dimen@\relax>.01pt
        \PackageError{floatstest}{Room \the\dimen@, expected
          \the\@tempdima}\@ehc\fi
    \endgroup
  \fi}
\makeatother
\newcounter{note}

  \marginpar{\footnotesize NOTE\thenote: #1}\footnote{Foot \thenote.}}
