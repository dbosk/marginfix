% Test of notes beside bottom floats (\marginnotebesidefloats).
%
% A page with a table at the bottom and more footnotes than fit in the
% margin beside the text.  \floatcase chooses the page:
%
%   plain     a plain table with an ordinary caption and a \label;
%   ragged    the same with \raggedbottom;
%   wide      the table in adjustwidth with a negative right margin;
%   overfull  a table wider than the text, an overfull line;
%   left      the table in adjustwidth with a negative left margin, with
%             the notes in the left margin (\reversemarginpar);
%   verso     the table in adjustwidth* on a verso page, sticking out on
%             the outer (left) side, where the notes are;
%   side      the table with memoir's side caption;
%   marker    the plain table with \marginnotereservefloat;
%   two       two plain tables at the bottom;
%   topbot    a plain table at the top and one at the bottom;
%   block     the plain table, and the margin blocked across the end of
%             its page;
%   split     the plain table, with deferrable notes in mode split;
%   deferred  the plain table, and right after it a table with
%             \marginnotereservefloat, which (with one bottom float per
%             page) lands at the bottom of the next page.
%
% \besidemode is the mode, reserve (default) or use.  The footnotes have
% labels at their marks and in their texts, and at the end of the next run
% we check from the .aux file: with reserve, the plain table must still
% push a footnote to the next page (the problem); with use, the plain,
% ragged and split pages must not.  In mode use we also check, on every
% page with bottom floats that keep the margin free, that the room added
% beside them is exactly what LaTeX took from the text for them
% (\@colht-\@colroom, less what the top floats took).  In case deferred
% the marked table must land on a later page than its callout, and keep
% the margin beside it empty.  That the tables
% that extend into the margin keep the margin beside them empty is checked
% by test/floatsmode.sh, which compares the PDFs of the two modes.
% Needs three runs.

% Check the room beside the bottom floats against LaTeX's own account.
\AtBeginDocument{%
  \ifdefined\MFX@bottomroom
    \let\ft@bottomroom\MFX@bottomroom
    \def\MFX@bottomroom{\ft@bottomroom\ft@checkroom}%
  \fi}
\def\ft@checkroom{%
  \if@tempswa
    \global\advance\ft@wide\@ne
    \typeout{floatstest: page \thepage: a bottom float is wide}%
  \else
    \begingroup
      \@tempdima\dimexpr\@colht-\@colroom\relax
      \ifx\@toplist\@empty\else
        \@tempdimb\textfloatsep
        \advance\@tempdimb-\floatsep
        \def\@elt##1{%
          \advance\@tempdimb\dimexpr\ht##1+\dp##1+\floatsep\relax}%
        \@toplist
        \advance\@tempdima-\@tempdimb
      \fi
      \typeout{floatstest: page \thepage: room \the\dimen@,
        expected \the\@tempdima}%
      \ifdim\dimexpr\dimen@-\@tempdima\relax>.01pt
        \PackageError{floatstest}{Room beside bottom floats is
          \the\dimen@, expected \the\@tempdima}\@ehc
      \fi
      \ifdim\dimexpr\@tempdima-\dimen@\relax>.01pt
        \PackageError{floatstest}{Room beside bottom floats is
          \the\dimen@, expected \the\@tempdima}\@ehc
      \fi
    \endgroup
  \fi}
\AtEndDocument{%
  \ifnum\ft@page{fn\the\c@fnmark}<\z@
    \typeout{floatstest: labels undefined, rerun to check}%
  \else
    \ft@i\z@
    \@whilenum\ft@i<\c@fnmark\do{%
      \advance\ft@i\@ne
      \ifnum\ft@page{mk\the\ft@i}=\ft@page{fn\the\ft@i}\else
        \advance\ft@off\@ne
      \fi}%
    \typeout{floatstest (\floatcase, \besidemode): \the\c@fnmark\space
      footnotes, \the\ft@off\space off their mark's page}%
    \def\reserved@a{use}%
    \ifx\besidemode\reserved@a
      \@tempswafalse
      \IfCase{plain}{\@tempswatrue}{}%
      \IfCase{ragged}{\@tempswatrue}{}%
      \IfCase{split}{\@tempswatrue}{}%
      \if@tempswa\ifnum\ft@off>\z@
        \PackageError{floatstest}{Footnotes moved beside a narrow bottom
          float in mode use}\@ehc
      \fi\fi
    \else
      \IfCase{plain}{\ifnum\ft@off=\z@
        \PackageError{floatstest}{No footnote moved in mode reserve:
          the test no longer reproduces the problem}\@ehc
      \fi}{}%
    \fi
    \IfCase{deferred}{%
      \ifnum\ft@page{tabm}>\ft@page{tabcall}\else
        \PackageError{floatstest}{The marked table was not deferred}\@ehc
      \fi
      \ifx\besidemode\reserved@a\ifnum\ft@wide=\z@
        \PackageError{floatstest}{The deferred marked table did not keep
          the margin beside it empty}\@ehc
      \fi\fi}{}%
  \fi}

\newcommand\testtable{%
  \begin{table}[b]
    \IfCase{marker}{\marginnotereservefloat}{}%
    \IfCase{side}{%
      \begin{sidecaption}{A table with a side caption.}[tab]
        \plaintable
      \end{sidecaption}}{%
    \IfCase{wide}{%
      \begin{adjustwidth}{0pt}{-4cm}\plaintable\end{adjustwidth}
      \caption{A table wider than the column.}\label{tab}}{%
    \IfCase{left}{%
      \begin{adjustwidth}{-4cm}{0pt}\plaintable\end{adjustwidth}
      \caption{A table sticking out on the left.}\label{tab}}{%
    \IfCase{verso}{%
      \begin{adjustwidth*}{0pt}{-4cm}\plaintable\end{adjustwidth*}
      \caption{A table sticking out on the outer side.}\label{tab}}{%
    \IfCase{overfull}{%
      \centering\begin{tabular}{p{7cm}p{7cm}l}\rows\end{tabular}
      \caption{A table wider than the text.}\label{tab}}{%
      \plaintable
      \caption{A plain table at the bottom.}\label{tab}}}}}}%
  \end{table}%
  \IfCase{two}{%
    \begin{table}[b]
      \centering\begin{tabular}{lll}\hline X & Y & Z\\ \hline\end{tabular}
      \caption{A second table at the bottom.}
    \end{table}}{}%
  \IfCase{topbot}{%
    \begin{table}[t]
      \centering\begin{tabular}{lll}\hline X & Y & Z\\ 1 & 2 & 3\\
        \hline\end{tabular}
      \caption{A table at the top.}
    \end{table}}{}}
