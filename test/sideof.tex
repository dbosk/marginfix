% Test of \marginparsideof{<page>}, which says (in \ifmarginparleft) in which
% margin marginfix sets the notes of a page, as \MFX@marginside decides it,
% for a package that needs to know (didactic places things beside notes).
% \sideofclass and \sideofopts choose the document class and its options,
% \sideofmargin memoir's \marginparmargin, \sideofreverse \reversemarginpar.
% The expected sides of pages 1 and 2 are \sideofwant (two letters, l or r);
% test/sideofcheck.sh runs the cases.  It is no test of the layout, so the
% regression test skips it.
% regress: skip
\providecommand\sideofclass{memoir}
\providecommand\sideofopts{twoside}
\expandafter\documentclass\expandafter[\sideofopts]{\sideofclass}
\usepackage{marginfix}
\ifdefined\sideofmargin\expandafter\marginparmargin\expandafter{\sideofmargin}\fi
\ifdefined\sideofreverse\reversemarginpar\fi
\makeatletter
\def\sideof@one#1{\marginparsideof{#1}%
  \edef\sideof@got{\sideof@got\ifmarginparleft l\else r\fi}}
\def\sideof@check{%
  \let\sideof@got\@empty \sideof@one1\sideof@one2%
  \ifx\sideof@got\sideofwant
    \typeout{sideof test (\sideofclass, \sideofopts): pages 1 and 2: \sideof@got}%
  \else
    \PackageError{sideof test}{Pages 1 and 2 got \sideof@got, not \sideofwant}{}%
  \fi}
\makeatother
\begin{document}
\makeatletter\sideof@check\makeatother
Text.
\end{document}
