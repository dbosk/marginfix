% Test of records that a page no longer needs (\marginnoteoverflow{break}).
%
% Two paragraphs dense with long notes in the margin, the second further on.
% A build from no .aux file records, in the same run, the first note that
% doesn't fit on each page, each from the layout before any page ends
% early.  Once the first paragraph's page ends early, the text after it
% moves, and the second record can end a page a few lines from its top,
% where its note would fit.  \stalecase chooses:
%
%   stale    memoir, its footnotes in the margin: the second record must be
%            retired, so that no page ends early for a note that fits, and
%            the page fills;
%   article  the same in the article class, with \marginpar;
%   needed   memoir, with more space before the first paragraph: both
%            records are needed (each page that ends early is full of
%            notes), and must stay, though one page is less than half full;
%   linemate stale, with a long note called out on the same line just
%            before the note of the second record: the page has room for
%            that note alone, but not for both, which come to it together,
%            so the record is needed and must not be retired, not even
%            once.
%
% \stalepriority, if defined, is a \marginnotepriority mode, and \stalelimit a
% \marginnoteyieldlimit in lines: the stale record must be retired under each
% of them too, and under a limit its page must not be capped for it.
%
% \stalelie, if defined, is the key of a note whose height is written wrong
% (1pt) to the .aux file at the end of each run, after marginfix's own record
% of it, so that the check of its page is wrong in every run: the runs must
% settle all the same, with the page ending early for the note.
%
% test/stalecheck.sh runs the cases and checks them.
\def\stalememoir{stale}\def\stalearticlecase{article}
\def\staleneeded{needed}\def\stalelinemate{linemate}
\ifx\stalecase\stalearticlecase
  \documentclass[a4paper]{article}
\else
  \documentclass[a4paper]{memoir}
\fi
\usepackage{lipsum}
\usepackage{marginfix}
\marginnoteoverflow{break}
\ifdefined\stalepriority\marginnotepriority{\stalepriority}\fi
\ifdefined\stalelimit\marginnoteyieldlimit{\stalelimit\baselineskip}\fi
\ifx\stalecase\stalearticlecase
  \newcommand\fn[1]{\marginpar{\footnotesize #1 \lipsum[1][1-2]}}
\else
  \footnotesinmargin
  \newcommand\fn[1]{\footnote{#1 \lipsum[1][1-5]}}
\fi
\ifx\stalecase\staleneeded
  \def\staleA{16}\def\staleB{12}
\else
  \def\staleA{0}\def\staleB{24}
\fi
\ifdefined\stalelie
  \makeatletter
  \AtEndDocument{\immediate\write\@auxout{\string\mfx@yieldht{\stalelie}{1pt}}}
  \makeatother
\fi
\ifx\stalecase\stalelinemate
  \def\matefn{\fn{X \lipsum[1][1-10]}}
\else
  \def\matefn{}
\fi
\begin{document}
\lipsum[1-3]

\vspace*{\staleA\baselineskip}

\lipsum[5][1-3]\fn{D} \lipsum[6][1-3]\fn{E} \lipsum[7][1-3]\fn{F}
\lipsum[8][1-3]\fn{G}\par

\lipsum[13-14]

\vspace*{\staleB\baselineskip}
\lipsum[9][1-3]\fn{H} \lipsum[10][1-3]\fn{I} \lipsum[11][1-3]\matefn\fn{J}
\lipsum[12][1-3]\fn{K} \lipsum[2][1-3]\fn{L} \lipsum[3][1-3]\fn{M}
\lipsum[4][1-3]\fn{N}\par

\lipsum[15-20]
\end{document}
