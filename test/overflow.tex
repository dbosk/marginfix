% Test of notes that don't fit on their page (\marginnoteoverflow).
%
% A paragraph with eleven long footnotes in the margin (memoir's
% \footnotesinmargin), two or three to a line, at the foot of the first
% page: more than its margin, and the next page's, can hold.
% \overflowcase chooses the mode:
%
%   carry  the default: the notes that don't fit are carried to the next
%          page, each logged as "marginfix-spill: ... intended";
%   break  \marginnoteoverflow{break} in the preamble: after the runs
%          settle, each page ends before the line of the first note that
%          doesn't fit, no note is carried (no marginfix-spill line), and
%          each early end is logged as "marginfix-yield: ...";
%   local  break in the preamble, but the paragraph in a group with
%          \marginnoteoverflow{carry}: as carry.
%   cap    break with \marginnoteyieldlimit{10\baselineskip}: the first
%          page ends a few lines early and keeps its break, the second
%          would end after a few lines, so after the runs settle it
%          doesn't, and its notes are carried, logged as
%          "marginfix-spill: ... capped" (none intended or unresolved),
%          each with a warning.
%   caphigh  break with a limit of 60 lines, more than a page: no note is
%          capped, as in break.
%   captall  as cap, with a line that is too tall: ending the page before it
%          leaves more than the limit, though its callout is low enough
%          (\overflowtall and \overflowtallnote below).  That note must
%          settle as capped, with a page that doesn't end early.
% \overflowgap, if defined, is a number of lines of space before the
% paragraph, so that the text moves while the notes keep their keys, as when
% the text before them changes in another file.  \overflowtall, if defined,
% is a number of lines that the line of note \overflowtallnote is high, so
% that the room above that line is much more than the room above its baseline.
%
% test/overflowcheck.sh runs each case and checks the log and that the
% runs settle.
\documentclass[a4paper]{memoir}
\usepackage{lipsum}
\usepackage{marginfix}
\footnotesinmargin
\def\breakcase{break}\def\localcase{local}\def\capcase{cap}
\def\caphighcase{caphigh}
\ifx\overflowcase\breakcase\marginnoteoverflow{break}\fi
\ifx\overflowcase\capcase
  \marginnoteoverflow{break}\marginnoteyieldlimit{10\baselineskip}\fi
\ifx\overflowcase\localcase\marginnoteoverflow{break}\fi
\ifx\overflowcase\caphighcase
  \marginnoteoverflow{break}\marginnoteyieldlimit{60\baselineskip}\fi
\newcommand\fn[1]{\ifdefined\overflowtall\ifnum\value{footnote}=\overflowtallnote
    \rule{0pt}{\overflowtall\baselineskip}\fi\fi
  \footnote{#1 \lipsum[1][1-6]}}
\begin{document}
\lipsum[1-4]

\ifdefined\overflowgap\vspace*{\overflowgap\baselineskip}\fi
\begingroup
\ifx\overflowcase\localcase\marginnoteoverflow{carry}\fi
\lipsum[2][1-3]\fn{A} \lipsum[3][1-3]\fn{B} \lipsum[4][1-3]\fn{C}
\lipsum[5][1-3]\fn{D} \lipsum[6][1-3]\fn{E} \lipsum[7][1-3]\fn{F}
\lipsum[8][1-3]\fn{G} \lipsum[9][1-3]\fn{H} \lipsum[10][1-3]\fn{I}
\lipsum[11][1-3]\fn{J} \lipsum[12][1-3]\fn{K}\par
\endgroup

\lipsum[6-8]
\end{document}
